cmake_minimum_required(VERSION 3.16)

project(Qqsp
    VERSION 1.9.0.0
    LANGUAGES C CXX
)

# Export the compilation database for clangd and other analysis tools.
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

# Match the language standards used by the current qmake project.
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)

# Qt code generation.
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
set(CMAKE_AUTOUIC ON)

option(QQSP_ENABLE_WEBBOX
    "Build the Qt WebEngine-based web box"
    ON
)

option(QQSP_ENABLE_WEBBOX_WEBKIT
    "Build the legacy Qt WebKit-based web box"
    OFF
)

option(QQSP_ENABLE_DEBUG
    "Enable the Qqsp memory-debugging support"
    OFF
)

option(QQSP_ENABLE_ANDROID
    "Enable Android-specific support"
    OFF
)

option(QQSP_IGNORE_QSP_INCOMPATIBLE_POINTER_WARNING
    "Ignore -Wincompatible-pointer-types warnings"
    ON
)

include(CheckTypeSize)

set(QQSP_ONIG_CONFIG_DIR "${CMAKE_CURRENT_BINARY_DIR}/qsp/onig")
set(QQSP_ONIG_CONFIG_TEMPLATE "${CMAKE_CURRENT_BINARY_DIR}/qsp/onig/config.h.in")

check_include_file("alloca.h" QQSP_ONIG_HAVE_ALLOCA_H)

check_include_file("string.h" QQSP_ONIG_HAVE_STRING_H)
check_include_file("sys/stat.h" QQSP_ONIG_HAVE_SYS_STAT_H)
check_include_file("sys/times.h" QQSP_ONIG_HAVE_SYS_TIMES_H)

check_include_file("sys/types.h" QQSP_ONIG_HAVE_SYS_TYPES_H)
check_include_file("unistd.h" QQSP_ONIG_HAVE_UNISTD_H)
check_include_files("sys/time.h;time.h" QQSP_ONIG_TIME_WITH_SYS_TIME)

check_type_size("int" QQSP_ONIG_SIZEOF_INT)
check_type_size("long" QQSP_ONIG_SIZEOF_LONG)
check_type_size("short" QQSP_ONIG_SIZEOF_SHORT)

set(QQSP_ONIG_HAVE_STDARG_PROTOTYPES 1)
set(QQSP_ONIG_STDC_HEADERS 1)

# link dependencies, so lower-level Qt/system libraries are not listed here.

    Widgets

        WebKitWidgets
    )
endif()

if(QQSP_ENABLE_ANDROID)
    if(NOT ANDROID)
        message(FATAL_ERROR
            "QQSP_ENABLE_ANDROID requires an Android Qt build."
        )
    endif()

    find_package(Qt5 REQUIRED COMPONENTS
        AndroidExtras
    )
endif()

if(QQSP_IGNORE_QSP_INCOMPATIBLE_POINTER_WARNING)
    target_compile_options(Qqsp
        PRIVATE
            $<$<COMPILE_LANGUAGE:C>:-Wno-incompatible-pointer-types>
    )
endif()

target_include_directories(Qqsp
    PRIVATE
        ${CMAKE_CURRENT_SOURCE_DIR}/qsp/bindings/default
        ${CMAKE_CURRENT_SOURCE_DIR}/qsp/onig
        ${QQSP_ONIG_CONFIG_DIR}
)

target_compile_definitions(Qqsp
    PRIVATE
        QT_DEPRECATED_WARNINGS
        QT_DISABLE_DEPRECATED_BEFORE=0x060000
        _UNICODE
        NOT_RUBY
)

if(QQSP_ENABLE_DEBUG)
    target_compile_definitions(Qqsp
        PRIVATE
            _DEBUGMW
    )
endif()

if(QQSP_ENABLE_WEBBOX)
    target_compile_definitions(Qqsp
        PRIVATE
            _WEBBOX
            _WEBBOX_COMMON
    )
endif()

if(QQSP_ENABLE_WEBBOX_WEBKIT)
    target_compile_definitions(Qqsp
        PRIVATE
            _WEBBOX_WEBKIT
            _WEBBOX_COMMON
    )
endif()

if(QQSP_ENABLE_ANDROID)
    target_compile_definitions(Qqsp
        PRIVATE
            _ANDROIDQT
    )
endif()

# Match qmake's unix:!macx disable-nativedialog configuration.

    target_compile_definitions(Qqsp

# Preserve the qmake application bundle/icon behavior on macOS.
if(APPLE)
    set_target_properties(Qqsp PROPERTIES
        MACOSX_BUNDLE TRUE
        MACOSX_BUNDLE_ICON_FILE qsp-logo.icns
    )

    target_sources(Qqsp
        PRIVATE
            icons/qsp-logo.icns
    )

    set_source_files_properties(
        icons/qsp-logo.icns
        PROPERTIES
            MACOSX_PACKAGE_LOCATION Resources
    )
endif()

# Qqsp is a GUI subsystem application on Windows.
if(WIN32)
    set_target_properties(Qqsp PROPERTIES
        WIN32_EXECUTABLE TRUE
    )
endif()
